# librsvg.wasm - Production-Ready SVG Rendering via WASM
# Dual build architecture: SIDE_MODULE + MAIN_MODULE (NPM)
#
# LGPL-2.1-or-later License: Permits dynamic linking in commercial products
# SVG rendering with WebGPU acceleration and SIMD optimizations
#
# Based on librsvg 2.61.0 with maximum browser feature utilization
# Targets Chrome/Edge 113+ exclusively (WebGPU + WASM SIMD required)

cmake_minimum_required(VERSION 3.22)

# Project configuration
project(LibRSVG_WASM
    VERSION 2.61.0
    LANGUAGES C CXX
    DESCRIPTION "Production SVG rendering library with WebGPU acceleration"
)

# Build architecture selector
option(BUILD_SIDE_MODULE "Build as SIDE_MODULE" OFF)
option(BUILD_MAIN_MODULE "Build as MAIN_MODULE for NPM distribution" ON)

# Performance and feature options
option(ENABLE_SIMD "Enable WASM SIMD optimizations" ON)
option(ENABLE_WEBGPU "Enable WebGPU compute acceleration" ON)
option(ENABLE_THREADING "Enable SharedArrayBuffer threading" ON)
option(ENABLE_LTO "Enable link-time optimization" ON)
option(ENABLE_ASYNCIFY "Enable async/await in C/C++ code" ON)

# WASM core compiler flags - Maximum performance configuration
set(WASM_CORE_FLAGS
    "-s WASM=1"
    "-s MODULARIZE=1"
    "-s EXPORT_ES6=1"
    "-O3"                         # Maximum optimization
    "-flto"                       # Link-time optimization
    "--closure=1"                 # Google Closure Compiler
    "-s ALLOW_MEMORY_GROWTH=1"
    "-s DISABLE_EXCEPTION_CATCHING=0"
    "-s ASSERTIONS=0"             # Remove assertions for production
)

# Memory configuration - 4GB maximum for professional SVG processing
set(WASM_MEMORY_FLAGS
    "-s INITIAL_MEMORY=134217728"   # 128MB initial
    "-s MAXIMUM_MEMORY=4294967296"  # 4GB maximum (professional workflow support)
    "-s STACK_SIZE=4194304"         # 4MB stack (complex SVG parsing)
    "-s ALLOW_MEMORY_GROWTH=1"
    "-s WASM_MEM_MAX=4294967296"    # Explicit 4GB limit
)

# Threading and SharedArrayBuffer (Chrome/Edge 113+)
if(ENABLE_THREADING)
    list(APPEND WASM_CORE_FLAGS
        "-s USE_PTHREADS=1"
        "-s PTHREAD_POOL_SIZE=8"        # 8 worker threads
        "-s PTHREAD_POOL_SIZE_STRICT=2" # Always keep 2 threads alive
        "-s SHARED_MEMORY=1"
        "-s PROXY_TO_PTHREAD=1"         # Main thread stays responsive
    )
endif()

# WASM SIMD optimizations (Chrome 91+, mandatory for native performance)
if(ENABLE_SIMD)
    list(APPEND WASM_CORE_FLAGS
        "-msimd128"
        "-s SIMD=1"
    )
    # SVG-specific SIMD optimizations
    add_compile_definitions(LIBRSVG_SIMD_ENABLED=1)
endif()

# WebGPU compute acceleration (Chrome 113+, future)
if(ENABLE_WEBGPU)
    list(APPEND WASM_CORE_FLAGS
        "-s USE_WEBGPU=1"
    )
    add_compile_definitions(LIBRSVG_WEBGPU_ENABLED=1)
endif()

# Asyncify for async/await support in C/C++
if(ENABLE_ASYNCIFY)
    list(APPEND WASM_CORE_FLAGS
        "-s ASYNCIFY=1"
        "-s ASYNCIFY_STACK_SIZE=32768"
        # SVG parsing functions that may need async behavior
        "-s ASYNCIFY_ALLOWLIST=['rsvg_handle_new_from_data','rsvg_handle_render_cairo']"
    )
endif()

# Advanced optimization flags (link-time only)
list(APPEND WASM_CORE_FLAGS
    "-s ELIMINATE_DUPLICATE_FUNCTIONS=1"
    "-s MERGE_FUNCTIONS=1"
    "-s AGGRESSIVE_VARIABLE_ELIMINATION=1"
)

# SVG-specific exported functions for TypeScript integration
set(WASM_EXPORTED_FUNCTIONS
    '_malloc'
    '_free'
    # Core librsvg API
    '_rsvg_handle_new'
    '_rsvg_handle_new_from_data'
    '_rsvg_handle_set_dpi'
    '_rsvg_handle_get_dimensions'
    '_rsvg_handle_render_cairo'
    '_rsvg_handle_close'
    # Cairo integration
    '_cairo_image_surface_create'
    '_cairo_image_surface_get_data'
    '_cairo_image_surface_get_width'
    '_cairo_image_surface_get_height'
    '_cairo_image_surface_get_stride'
    '_cairo_create'
    '_cairo_destroy'
    '_cairo_surface_destroy'
    # WASM-native SVG functions (custom additions)
    '_rsvg_wasm_render_to_rgba'
    '_rsvg_wasm_get_svg_info'
    '_rsvg_wasm_set_viewport'
    '_rsvg_wasm_get_simd_support'
    '_rsvg_wasm_enable_webgpu'
    '_rsvg_wasm_get_last_error'
    '_rsvg_wasm_cleanup'
)

# Convert exported functions to Emscripten format
string(REPLACE ";" "\",\"" EXPORTED_FUNCTIONS_STR "${WASM_EXPORTED_FUNCTIONS}")
set(EXPORTED_FUNCTIONS_STR "[\"${EXPORTED_FUNCTIONS_STR}\"]")

list(APPEND WASM_CORE_FLAGS
    "-s"
    "EXPORTED_FUNCTIONS=${EXPORTED_FUNCTIONS_STR}"
    "-s"
    "EXPORTED_RUNTIME_METHODS=[\"ccall\",\"cwrap\",\"UTF8ToString\",\"stringToUTF8\"]"
)

# Build configuration selection
if(BUILD_SIDE_MODULE)
    # SIDE_MODULE: For dynamic loading into main app
    message(STATUS "Building SIDE_MODULE")

    list(APPEND WASM_CORE_FLAGS
        "-s SIDE_MODULE=1"
        "-fPIC"                       # Position-independent code
    )

    set(OUTPUT_NAME "librsvg-side")
    add_compile_definitions(LIBRSVG_SIDE_MODULE=1)

elseif(BUILD_MAIN_MODULE)
    # MAIN_MODULE: Self-contained NPM package
    message(STATUS "Building MAIN_MODULE for NPM distribution")

    list(APPEND WASM_CORE_FLAGS
        "-s MAIN_MODULE=1"
        "-s EXPORT_ALL=1"             # Export all symbols for testing
        "-s INVOKE_RUN=0"             # Don't auto-run main()
        "-s EXIT_RUNTIME=0"           # Keep runtime alive
    )

    set(OUTPUT_NAME "librsvg-main")
    add_compile_definitions(LIBRSVG_MAIN_MODULE=1)

else()
    message(FATAL_ERROR "Must specify either BUILD_SIDE_MODULE=ON or BUILD_MAIN_MODULE=ON")
endif()

# Compiler setup
if(NOT CMAKE_C_COMPILER MATCHES "emcc")
    message(FATAL_ERROR "This project requires Emscripten. Use: emcmake cmake ...")
endif()

# Source files (librsvg core + WASM integration layer)
set(LIBRSVG_SOURCES
    # Core librsvg Rust sources are built via Cargo
    # We only include the WASM integration layer here
    wasm/librsvg-wasm-integration.c
    wasm/librsvg-wasm-simd-optimizations.c
)

# Create the WASM library target
add_library(${OUTPUT_NAME} ${LIBRSVG_SOURCES})

# Apply compiler flags (only non-Emscripten specific ones for compilation)
target_compile_options(${OUTPUT_NAME} PRIVATE
    "-O3"
    "-ffast-math"
    "-funroll-loops"
    "-fomit-frame-pointer"
)

# Apply SIMD flags during compilation if enabled
if(ENABLE_SIMD)
    target_compile_options(${OUTPUT_NAME} PRIVATE "-msimd128")
endif()

# Apply all WASM flags at link time only
target_link_options(${OUTPUT_NAME} PRIVATE ${WASM_CORE_FLAGS} ${WASM_MEMORY_FLAGS})

# Include directories
target_include_directories(${OUTPUT_NAME} PRIVATE
    include/
    rsvg/
    librsvg-c/
    wasm/
)

# Link with required libraries (system dependencies)
target_link_libraries(${OUTPUT_NAME}
    # Note: Rust librsvg core is linked via rustc, not CMake
    # These are additional system dependencies for WASM
)

# Set output properties
set_target_properties(${OUTPUT_NAME} PROPERTIES
    OUTPUT_NAME ${OUTPUT_NAME}
    SUFFIX ".wasm"
)

# Installation rules
install(TARGETS ${OUTPUT_NAME}
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

# Generate TypeScript definition files during build
add_custom_command(TARGET ${OUTPUT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E echo "WASM build completed: ${OUTPUT_NAME}.wasm"
    COMMAND ${CMAKE_COMMAND} -E echo "Size: $$(wc -c < ${OUTPUT_NAME}.wasm) bytes"
    COMMAND ${CMAKE_COMMAND} -E echo "Exported functions: ${WASM_EXPORTED_FUNCTIONS}"
)

# Development targets
add_custom_target(info
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "librsvg.wasm Build Information:"
    COMMAND ${CMAKE_COMMAND} -E echo "================================"
    COMMAND ${CMAKE_COMMAND} -E echo "Version: ${PROJECT_VERSION}"
    COMMAND ${CMAKE_COMMAND} -E echo "Build Type: $<IF:$<BOOL:${BUILD_SIDE_MODULE}>,SIDE_MODULE,MAIN_MODULE>"
    COMMAND ${CMAKE_COMMAND} -E echo "SIMD: $<IF:$<BOOL:${ENABLE_SIMD}>,Enabled,Disabled>"
    COMMAND ${CMAKE_COMMAND} -E echo "WebGPU: $<IF:$<BOOL:${ENABLE_WEBGPU}>,Enabled,Disabled>"
    COMMAND ${CMAKE_COMMAND} -E echo "Threading: $<IF:$<BOOL:${ENABLE_THREADING}>,Enabled,Disabled>"
    COMMAND ${CMAKE_COMMAND} -E echo "LTO: $<IF:$<BOOL:${ENABLE_LTO}>,Enabled,Disabled>"
    COMMAND ${CMAKE_COMMAND} -E echo "Asyncify: $<IF:$<BOOL:${ENABLE_ASYNCIFY}>,Enabled,Disabled>"
    COMMAND ${CMAKE_COMMAND} -E echo "Target: Chrome/Edge 113+ (WebGPU+SIMD required)"
    COMMAND ${CMAKE_COMMAND} -E echo "Memory: 128MB initial, 4GB maximum"
    COMMAND ${CMAKE_COMMAND} -E echo "Threads: 8 workers (2 persistent)"
    COMMAND ${CMAKE_COMMAND} -E echo ""
)

# Print configuration summary
message(STATUS "")
message(STATUS "librsvg.wasm Configuration Summary:")
message(STATUS "===================================")
message(STATUS "Version: ${PROJECT_VERSION}")
message(STATUS "Build Type: ${CMAKE_BUILD_TYPE}")
message(STATUS "Module Type: $<IF:$<BOOL:${BUILD_SIDE_MODULE}>,SIDE_MODULE,MAIN_MODULE>")
message(STATUS "SIMD: ${ENABLE_SIMD}")
message(STATUS "WebGPU: ${ENABLE_WEBGPU}")
message(STATUS "Threading: ${ENABLE_THREADING}")
message(STATUS "LTO: ${ENABLE_LTO}")
message(STATUS "Asyncify: ${ENABLE_ASYNCIFY}")
message(STATUS "Max Memory: 4GB")
message(STATUS "Target: Chrome/Edge 113+")
message(STATUS "Output: ${OUTPUT_NAME}.wasm")
message(STATUS "")
